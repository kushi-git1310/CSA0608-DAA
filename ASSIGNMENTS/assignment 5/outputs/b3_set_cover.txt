  Greedy trace  :
    pick C3 (covers 6 new zones), uncovered before = 9
    pick C1 (covers 2 new zones), uncovered before = 3
    pick C2 (covers 1 new zones), uncovered before = 1
  Greedy cover  : ['C3', 'C1', 'C2'] -> size 3
  GreedyCoverSize (3) > OptimalCoverSize (2)
